const BYTES = Buffer.from('FICTIONAL-RECONCILE-CONTENT');
function fixture() {
  const root = mkdtempSync(join(tmpdir(), 'psc-claude-reconcile-'));
  const dataDir = join(root, 'private');
  const keys = loadPrivateKeys(join(root, 'keys'), { allowCreate: true });
  const accounts = createAccountStore(join(root, 'auth.sqlite'));
  const owner = accounts.create({ email: 'owner@example.test', passwordHash: 'x' }).id;
  const boot = (fileOps?: Partial<PrivateFileOps>) => {
    const store = createPrivateStore({ db: accounts.db, keys, dataDir, fileOps });
    return { store, report: store.reconcile() };
  };
  const first = boot().store;
  const caseId = first.createCase(owner, { label: 'Fictional', jurisdiction: null }).id;
  const docId = first.createDocument(owner, caseId, { originalName: 'f.pdf', mimeType: 'application/pdf', bytes: BYTES,
    sha256: createHash('sha256').update(BYTES).digest('hex') }).id;
  const files = (dir: string): string[] => !existsSync(dir) ? [] : readdirSync(dir, { withFileTypes: true })

  const done = () => { accounts.close(); rmSync(root, { recursive: true, force: true }); };
  return { root, dataDir, owner, caseId, docId, boot, files, canOpen, done };
}
